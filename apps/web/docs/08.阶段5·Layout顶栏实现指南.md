## 步骤 2：新建主题色工具 `src/utils/theme.ts`

Element Plus 的主色依赖 `--el-color-primary` 以及 6 个衍生变量（`light-3/5/7/8/9`、`dark-2`）。只改主色不够，必须同步写入这些变体，按钮/激活态才会整体变色。新建 `src/utils/theme.ts`：

```ts
// src/utils/theme.ts
// 主题色工具：把选中的主色写入根元素的 CSS 变量，驱动 Element Plus 整体换肤

const PRIMARY_COLOR_KEY = 'go-vue-selection-primary-color'

// 统一解析任意颜色为 rgb 元组；支持 #fff / #ffffff / rgb(...) / rgba(...)
export function parseColor(color: string): [number, number, number] {
  const trimmed = color.trim()
  if (trimmed.startsWith('#')) {
    let normalized = trimmed.slice(1)
    if (normalized.length === 3) {
      normalized = normalized.split('').map((c) => c + c).join('')
    }
    const num = parseInt(normalized, 16)
    return [(num >> 16) & 255, (num >> 8) & 255, num & 255]
  }
  const rgbMatch = trimmed.match(/rgba?\s*\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/)
  if (rgbMatch) {
    return [parseInt(rgbMatch[1], 10), parseInt(rgbMatch[2], 10), parseInt(rgbMatch[3], 10)]
  }
  throw new Error(`Unsupported color format: ${color}`)
}

// 把 rgb 三元组转回 hex 字符串
export function rgbToHex(r: number, g: number, b: number): string {
  const toHex = (n: number) => n.toString(16).padStart(2, '0')
  return `#${toHex(r)}${toHex(g)}${toHex(b)}`
}

// 颜色混合：把 color1 与 color2 按比例 weight 混合（weight 越大越接近 color2）
export function mix(color1: string, color2: string, weight: number): string {
  const [r1, g1, b1] = parseColor(color1)
  const [r2, g2, b2] = parseColor(color2)
  const r = Math.round(r1 * (1 - weight) + r2 * weight)
  const g = Math.round(g1 * (1 - weight) + g2 * weight)
  const b = Math.round(b1 * (1 - weight) + b2 * weight)
  return rgbToHex(r, g, b)
}

// 把主色写入根元素：Element Plus 的浅色/深色变体全部由主色推导
export function setPrimaryColor(primary: string, el: HTMLElement = document.documentElement): void {
  let normalized: string
  try {
    normalized = rgbToHex(...parseColor(primary))
  } catch {
    normalized = '#409eff'
  }

  el.style.setProperty('--el-color-primary', normalized)
  // EP 组件内部用 rgba(var(--el-color-primary-rgb), x) 做半透明效果，必须是 r, g, b 数字串
  const [r, g, b] = parseColor(normalized)
  el.style.setProperty('--el-color-primary-rgb', `${r}, ${g}, ${b}`)

  // 浅色变体：与主色混合白色
  const lightWeights: Record<string, number> = {
    'light-3': 0.3,
    'light-5': 0.5,
    'light-7': 0.7,
    'light-8': 0.8,
    'light-9': 0.9,
  }
  for (const [key, weight] of Object.entries(lightWeights)) {
    el.style.setProperty(`--el-color-primary-${key}`, mix(normalized, '#ffffff', weight))
  }
  // 深色变体：与主色混合黑色
  el.style.setProperty('--el-color-primary-dark-2', mix(normalized, '#000000', 0.2))

  // Sass 按需加载不会注入 dist/index.css 中 :root 的派生变量映射，浏览器回退到默认 #409eff，
  // 因此运行时切换主题色必须显式覆盖这些派生变量（hover/浅底用 light-9，加深用 dark-2）
  const light9 = mix(normalized, '#ffffff', 0.9)
  const dark2 = mix(normalized, '#000000', 0.2)
  const derivedVariables: Record<string, string> = {
    '--el-menu-active-color': normalized,
    '--el-menu-hover-text-color': normalized,
    '--el-menu-hover-bg-color': light9,
    '--el-button-primary-bg-color': normalized,
    '--el-button-primary-border-color': normalized,
    '--el-button-primary-hover-bg-color': dark2,
    '--el-button-primary-hover-border-color': dark2,
    '--el-button-primary-active-bg-color': dark2,
    '--el-button-primary-active-border-color': dark2,
    '--el-input-focus-border-color': normalized,
    '--el-input-hover-border-color': normalized,
    '--el-link-text-color': normalized,
    '--el-link-hover-text-color': dark2,
    '--el-radio-button-checked-bg-color': normalized,
    '--el-radio-button-checked-border-color': normalized,
    '--el-tabs-active-color': normalized,
    '--el-tabs-hover-color': normalized,
    '--el-switch-on-color': normalized,
    '--el-slider-main-bg-color': normalized,
    '--el-pagination-hover-color': normalized,
    '--el-checkbox-checked-input-border-color': normalized,
    '--el-checkbox-checked-bg-color': normalized,
    '--el-select-input-focus-border-color': normalized,
    '--el-tree-node-hover-bg-color': light9,
    '--el-tag-bg-color': light9,
    '--el-tag-border-color': normalized,
    '--el-tag-hover-color': dark2,
  }
  for (const [variable, value] of Object.entries(derivedVariables)) {
    el.style.setProperty(variable, value)
  }

  // 持久化
  localStorage.setItem(PRIMARY_COLOR_KEY, normalized)
}

// 读取持久化的主题色；没有则返回默认 EP 蓝
export function getStoredPrimaryColor(): string {
  return localStorage.getItem(PRIMARY_COLOR_KEY) || '#409eff'
}

// 应用持久化主题色（一般在 main.ts 应用启动时调用，避免刷新后闪回默认蓝）
export function applyStoredPrimaryColor(el: HTMLElement = document.documentElement): string {
  const color = getStoredPrimaryColor()
  setPrimaryColor(color, el)
  return color
}
```

> `mix` 是纯函数（无副作用），只做颜色混合；`setPrimaryColor` 把主色、变体、派生变量一并写进根元素，并持久化到 `localStorage`。后续任何组件想换主题色，调 `setPrimaryColor(color)` 即可。`applyStoredPrimaryColor()` 在应用启动时调用，恢复用户上次选的主题色。

## 步骤 3：修改 `main.ts` 引入暗黑样式

Element Plus 的暗黑模式需要加载它自带的暗黑 CSS 变量文件，再给 `<html>` 加 `dark` 类即可全局生效。在 `src/main.ts` **顶部**加一行引入：

```ts
// src/main.ts（只改最上面几行，其余不动）
import { createPinia } from 'pinia'
import { createApp } from 'vue'
import SvgIcon from '~virtual/svg-component' // 生成式全局图标组件
import App from './App.vue'
import router from './router'
import './style.css' // 自定义样式（放最后）
// ↑ 暗黑变量必须放在自定义 style.css 之前加载，否则会被覆盖

// 新增：Element Plus 暗黑模式变量（放在组件样式之前）
import 'element-plus/theme-chalk/dark/css-vars.css'

// 应用启动时先恢复用户持久化的主题色，避免刷新后闪回默认蓝
import { applyStoredPrimaryColor } from '@/utils/theme'

applyStoredPrimaryColor()

const app = createApp(App)
app.component('SvgIcon', SvgIcon)
app.use(createPinia())
app.use(router)
app.mount('#app')
```

> 必须放在自定义 `style.css` **之前**引入，确保暗黑变量不被业务样式覆盖。

Element Plus 暗黑只管它自己的组件（菜单/卡片/按钮…），**自定义背景（顶栏、主内容区、首页）不会自动变深**，需要手动写 `html.dark` 覆盖。先在全局 `src/style.css` 给页面背景兜底：

```css
/* src/style.css */
html,
body,
#app {
  width: 100%;
  height: 100%;
}

/* 暗黑模式下兜底页面背景（自定义区域之外的缝隙） */
html.dark body {
  background: #000;
}
```

各布局/页面组件内的具体覆盖见步骤 7.2 与 7.3。

## 步骤 4：面包屑组件 `src/layout/tabbar/breadcrumb/index.vue`

新建目录与文件 `src/layout/tabbar/breadcrumb/index.vue`：

```vue
<template>
  <el-breadcrumb separator="/">
    <el-breadcrumb-item v-for="item in matched" :key="item.path">
      <!-- 图标：meta.icon 是字符串，查 utils/icons.ts 的 routeIconMap 转组件（与左侧菜单同源） -->
      <el-icon v-if="getRouteIcon(item.meta?.icon)" class="layout-breadcrumb-icon">
        <component :is="getRouteIcon(item.meta?.icon)" />
      </el-icon>
      <!-- 纯位置指示：统一灰色、不可点击 -->
      <span class="layout-breadcrumb-text">{{ item.meta.title }}</span>
    </el-breadcrumb-item>
  </el-breadcrumb>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useRoute } from 'vue-router'
// 路由图标集中登记在 utils/icons.ts（见 03 文档步骤 4），面包屑与菜单共用同一套
import { getRouteIcon } from '@/utils/icons'

const $route = useRoute()

// route.matched 是当前路由匹配到的所有层级；过滤掉没 title / 隐藏的项
const matched = computed(() => $route.matched.filter((r) => r.meta?.title && !r.meta?.hidden))
</script>

<style scoped lang="scss">
.layout-breadcrumb-icon {
  margin-right: 4px;
  vertical-align: -2px;
}
.layout-breadcrumb-text {
  color: #606266;
}
</style>
```

> **定位**：面包屑只作「当前位置指示」用，所有项统一灰色、不可点击（不提供点击回退）。层级信息来自 `route.matched`，无需额外数据源。
>
> `route.matched` 对 `/home` 会得到 `[layout路由(无title), home子路由(title:首页)]`，过滤后只剩「首页」。访问 `/acl/user` 时（`routes.ts` 里 `/acl` 的 `children` 含 `path: 'user'`），`route.matched` 会得到 `[权限管理, 用户管理]`，两项都展示。

## 步骤 5：设置组件 `src/layout/tabbar/setting/index.vue`

新建 `src/layout/tabbar/setting/index.vue`，包含全屏、暗黑切换、主题色选择器：

```vue
<template>
  <div class="layout-tabbar-setting">
    <!-- 全屏：原生 Fullscreen API，不引入 screenfull -->
    <el-tooltip content="全屏" placement="bottom">
      <el-icon class="layout-setting-icon" @click="toggleFullscreen">
        <FullScreen />
      </el-icon>
    </el-tooltip>

    <!-- 暗黑模式：切 html.dark 类 -->
    <el-tooltip :content="isDark ? '亮色模式' : '暗黑模式'" placement="bottom">
      <el-icon class="layout-setting-icon" @click="toggleDark">
        <component :is="isDark ? Sunny : Moon" />
      </el-icon>
    </el-tooltip>

    <!-- 主题色：选色后写入 --el-color-primary 变体 -->
    <el-color-picker
      :model-value="primaryColor"
      :predefine="predefineColors"
      @change="changePrimary"
    />
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { FullScreen, Moon, Sunny } from '@element-plus/icons-vue'
import { getStoredPrimaryColor, setPrimaryColor } from '@/utils/theme'

// 用当前 DOM 上的 dark 类初始化，避免刷新后开关状态与页面不同步
const isDark = ref(document.documentElement.classList.contains('dark'))
// 初始化为持久化主题色，避免刷新后选择器显示与页面实际不一致
const primaryColor = ref(getStoredPrimaryColor())

// 主题色预设（点开选择器时快速选）
const predefineColors = ['#409EFF', '#1677ff', '#7265e6', '#28c76f', '#f5222d', '#fa541c']

const toggleFullscreen = () => {
  if (!document.fullscreenElement) {
    document.documentElement.requestFullscreen()
  } else {
    document.exitFullscreen()
  }
}

const toggleDark = () => {
  isDark.value = !isDark.value
  // 第二个参数强制设置类是否存在（而不是单纯 toggle），状态更可控
  document.documentElement.classList.toggle('dark', isDark.value)
}

const changePrimary = (color: string | null) => {
  if (!color) return
  primaryColor.value = color
  setPrimaryColor(color)
}
</script>

<style scoped lang="scss">
.layout-tabbar-setting {
  display: flex;
  align-items: center;
  gap: 18px;
}
.layout-setting-icon {
  font-size: 20px;
  cursor: pointer;
  color: #606266;
  transition: color 0.2s;
  &:hover {
    // 订阅主色变量：hover 时文字变主题色
    color: var(--el-color-primary);
  }
}
</style>
```

```vue
    border-bottom: 1px solid #303030;
    color: #e5eaf3;
  }
  .layout-main {
    background: #000;
  }
}
</style>
```

> 原来 `.layout-tabbar` 上的 `line-height` / `padding-left: 20px` 是给占位文本用的，现在内部 `Tabbar` 自己用 flex 布局，所以删掉，避免和内部 padding 冲突。

### 7.3 补充：首页等自定义页面的暗黑与主色适配

顶栏改完只是第一步。**自定义页面**同样需要暗黑覆盖与主色订阅，否则切暗黑后仍是白底、选主色后头像不变。

> 说明：下面以「首页」为例。**当前 `src/views/home/index.vue` 实际类名是 `.home-card / .greeting / .welcome-card / .welcome-illustration`，且文字色已用 `var(--el-text-color-secondary)` 等 EP 变量，暗黑下会自动适配，无需额外覆盖**。下方代码仅为「若你给首页加了 `.home` 等自定义类」时的写法示例，照抄前请先确认类名存在。

```scss
/* src/views/home/index.vue —— 暗黑模式覆盖（仅当页面存在对应类时才需要） */
html.dark {
  .home {
    background: #000;
  }
  .home-greeting-text {
    color: #e5eaf3;
  }
  .home-subtitle {
    color: #a3a6ad;
  }
  .home-card {
    box-shadow: 0 8px 24px rgb(0 0 0 / 20%);
  }
}

/* 头像底色订阅主色变量，选色后实时变化 */
.home-avatar {
  // 订阅主色变量：首页头像底色跟随主题色
  background: var(--el-color-primary);
}
```

> 原则：**写变量**（`setPrimaryColor` 写 `--el-color-primary`）+ **读变量**（自定义样式用 `var(--el-color-primary)`）= 完整换肤；暗黑同理，自定义背景/文字必须手写 `html.dark` 覆盖，EP 才不管。

### 7.4 进入页面时弹出问候通知

在 7.1 的 `src/layout/tabbar/index.vue` 基础上，把 `<script setup>` 升级为下面内容：进入布局（登录后）时弹出一个 `ElNotification` 问候通知。这里**手动 `import { ElNotification } from 'element-plus'`** 即可（与登录页写法一致），EP 会自行注入组件样式，无需依赖自动导入。

```ts
// src/layout/tabbar/index.vue —— <script setup> 升级版
import { computed, onMounted } from 'vue'
import { ElNotification } from 'element-plus'
import Breadcrumb from './breadcrumb/index.vue'
import Setting from './setting/index.vue'
import Userinfo from './userinfo/index.vue'
import { useUserStore } from '@/store/modules/user'

const userStore = useUserStore()

const displayName = computed(() => userStore.name ?? 'admin')

// 按当前小时计算问候语（与下拉里的逻辑一致）
const greeting = computed(() => {
  const hour = new Date().getHours()
  if (hour < 6) return '凌晨好'
  if (hour < 12) return '早上好'
  if (hour < 14) return '中午好'
  if (hour < 18) return '下午好'
  return '晚上好'
})

// 模块级标志：整页生命周期内只弹一次（避免 HMR / 重复挂载重复弹）
let notified = false
onMounted(() => {
  if (notified) return
  notified = true
  ElNotification({
    title: `Hi, ${greeting.value}!`,
    message: `欢迎回来，${displayName.value}`,
    type: 'success',
    duration: 3000,
    offset: 60,
  })
})
```

> 为什么用 `notified` 模块级变量而不是 `onMounted` 直接弹？开发模式下 Vite HMR 可能让组件模块重新求值，`onMounted` 每次挂载都会跑；用模块级 `let notified` 保证整个页面会话只弹一次。`ElNotification` 通过 `import { ElNotification } from 'element-plus'` 引入，组件样式由 EP 自动注入，写法与登录页一致。

## 附：学习要点与常见报错排查

### 学习要点

| 知识点                     | 在哪里用到                                             | 为什么重要                                                       |
| -------------------------- | ------------------------------------------------------ | ---------------------------------------------------------------- |
| **`route.matched` 面包屑** | `breadcrumb` 组件                                      | 路由自带层级信息，无需额外数据源即可做位置导航                   |
| **原生 Fullscreen API**    | `document.documentElement.requestFullscreen()`         | 零依赖实现全屏，避免为小功能引入 `screenfull`                    |
| **暗黑模式 `html.dark`**   | `main.ts` 引暗黑 css + `classList.toggle('dark')`      | Element Plus 官方暗黑方案，组件自动适配                          |
| **CSS 变量驱动主题色**     | `utils/theme.ts` 写 `--el-color-primary` 变体          | 一处改色，全局联动，是组件库换肤的标准做法                       |
| **自定义背景暗黑适配**     | `layout`/`style.css`/`home` 的 `html.dark` 覆盖        | EP 暗黑只管自身组件，自定义背景/文字需手动写 `html.dark` 覆盖    |
| **图标以组件对象传入**     | `setting` 组件 `<component :is="Sunny">`               | 顶栏功能图标（全屏/暗黑，非路由图标），直接 `import` 组件对象渲染 |
| **路由图标 `getRouteIcon`** | 面包屑/菜单 `meta.icon` → `utils/icons.ts` 的 `routeIconMap` | `meta.icon` 是字符串，经 `getRouteIcon` 查表转组件；新增图标须同步在 `utils/icons.ts` 登记（见 03 文档步骤 4） |
| **`el-color-picker`**      | 主题色选择器                                           | Element Plus 现成的颜色选择组件                                  |
| **`el-dropdown` 下拉**     | 用户头像区                                             | 后台常见的「头像 → 菜单」交互范式                                |
| **`ElNotification` 进入问候** | `tabbar/index.vue` 的 `onMounted` + 模块级 `notified` | 登录后弹通知；`import { ElNotification } from 'element-plus'`（与登录页一致），样式由 EP 自动注入 |

### 常见报错排查

| 报错                                          | 原因                                          | 解决                                                                                |
| --------------------------------------------- | --------------------------------------------- | ----------------------------------------------------------------------------------- |
| 暗黑模式切了没反应                            | `main.ts` 没引暗黑 css                        | 按步骤 3 在 `main.ts` 顶部加 `element-plus/theme-chalk/dark/css-vars.css` 引入      |
| 选了主题色但按钮/自定义区域不变色             | 只改了 `--el-color-primary`，漏了变体；或自定义样式硬编码 `#409eff` | 用 `setPrimaryColor()` 一次性写入全部变体；自定义处改成 `var(--el-color-primary)` 订阅主色变量 |
| 菜单 active 项（高亮文字）不变色，但按钮/头像变了 | `el-menu` 上写了 `active-text-color="#409eff"` 这个 props，EP 会生成内联样式顶掉 CSS 变量 `--el-menu-active-color` | 删掉 `el-menu` 的 `active-text-color` props，让它走 CSS 变量；或改成 `:active-text-color="getComputedStyle(document.documentElement).getPropertyValue('--el-color-primary')"` |
| 全屏报错 `requestFullscreen is not a function` | 在 iframe / 非用户手势中调用                  | 确保点击图标（用户手势）触发；本地 dev 正常                                         |
| 头像下拉不展开                                | `el-dropdown` 触发区非原生可聚焦元素          | 触发区用 `<span>`/`<div>` 包一层（本指南已处理）                                    |
| 暗黑模式下顶栏/主内容区仍白色                 | 自定义背景硬编码，未写 `html.dark` 覆盖       | 对应处加 `html.dark { .xxx { background: #000 } }`                                  |
| 暗黑图标不显示（unknown component）           | `<component :is="'Sunny'">` 传字符串          | 改为传组件对象 `<component :is="Sunny">`（先 import）                               |
| 面包屑/菜单图标空白（不报错）                 | `meta.icon` 写了名字却忘了在 `utils/icons.ts` 登记 | 在 `utils/icons.ts` 顶部 `import` 该图标并加进 `routeIconMap`（字符串名须与 `meta.icon` 一致，见 03 文档步骤 4） |
